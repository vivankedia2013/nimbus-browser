name: Build and release Nimbus
on:
  push:
    tags: ['v*']
  workflow_dispatch:
permissions:
  contents: write
jobs:
  build:
    strategy:
      fail-fast: false
      matrix:
        include:
          # macOS 11 to 26 (Intel + Apple Silicon) and Windows 10/11 use Electron 37
          - { name: mac,        os: macos-15,       args: '--mac --x64 --arm64', electron: '',        suffix: '' }
          # macOS 10.13 to 10.15 need the older Electron 22 (Intel only)
          - { name: mac-legacy, os: macos-15,       args: '--mac --x64',         electron: '22.3.27', suffix: '-legacy' }
          - { name: windows,    os: windows-latest, args: '--win --x64',         electron: '',        suffix: '' }
    runs-on: ${{ matrix.os }}
    defaults:
      run:
        shell: bash
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20 }
      - run: npm install
      - name: Use Electron ${{ matrix.electron }} for old macOS
        if: matrix.electron != ''
        run: npm install --no-save electron@${{ matrix.electron }}
      - name: Build installers
        run: npx electron-builder ${{ matrix.args }} --publish never "-c.artifactName=Nimbus-\${version}-\${os}-\${arch}${{ matrix.suffix }}.\${ext}"
      - uses: actions/upload-artifact@v4
        with:
          name: nimbus-${{ matrix.name }}
          path: |
            dist/*.dmg
            dist/*.exe
          if-no-files-found: error
  release:
    needs: build
    if: startsWith(github.ref, 'refs/tags/')
    runs-on: ubuntu-latest
    steps:
      - uses: actions/download-artifact@v4
        with:
          path: artifacts
          merge-multiple: true
      - uses: softprops/action-gh-release@v2
        with:
          name: Nimbus ${{ github.ref_name }}
          generate_release_notes: true
          body: |
            **Windows 10 / 11:** `Nimbus-*-win-x64.exe`
            **macOS 11 to 26, Apple Silicon (M1 and newer):** `Nimbus-*-mac-arm64.dmg`
            **macOS 11 to 26, Intel:** `Nimbus-*-mac-x64.dmg`
            **macOS 10.13 to 10.15 (Intel):** `Nimbus-*-mac-x64-legacy.dmg`

            Installers are not code-signed. Windows: *More info → Run anyway*. macOS: right-click the app → *Open*.
          files: artifacts/*
